function startDocumentExtraction() {
  const mainFolderId = '1RNjNwoa6shtlUMQC5LCF9aX_kckRiUwy'; // The ID of your root folder
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getActiveSheet();
  
  sheet.clear(); // Resets the sheet for a fresh sync
  sheet.appendRow(['Date', 'Session Name', 'Attendees', 'Month Folder', 'Doc Link']);
  
  const mainFolder = DriveApp.getFolderById(mainFolderId);
  processFolderRecursive(mainFolder, sheet);
}

function processFolderRecursive(folder, sheet) {
  const files = folder.getFiles();
  const folderName = folder.getName();

  while (files.hasNext()) {
    const file = files.next();
    const fileName = file.getName();
    
    // Regex: Captures (Date)_(Name)_(Attendees)
    const match = fileName.match(/^([0-9]{8})_(.*)_([0-9]+)$/);
    
    if (match) {
      // match[1] = Date, match[2] = Session, match[3] = Attendees
      sheet.appendRow([match[1], match[2], match[3], folderName, file.getUrl()]);
    }
  }

  // Dig into monthly subfolders
  const subfolders = folder.getFolders();
  while (subfolders.hasNext()) {
    processFolderRecursive(subfolders.next(), sheet);
  }
}
